# ---------------------------------------------------------------------------
# TITA thesis - workstation runtime dependencies
#
# Target: Ubuntu 22.04 / Python 3.10  (must match ROS 2 Humble's interpreter)
#   python3 -m venv --system-site-packages .venv
#   source .venv/bin/activate
#   pip install -r requirements.txt
#
# --system-site-packages is REQUIRED: rclpy, cv_bridge, tf2_ros and all other
# ROS 2 Python bindings come from apt (/opt/ros/humble), never from pip.
#
# On the robot (Jetson Orin NX) use requirements-jetson.txt instead - the
# CUDA-enabled wheels there come from NVIDIA's index, not from PyPI.
# ---------------------------------------------------------------------------

# --- Core / numerics -------------------------------------------------------
# numpy MUST stay <2.0: the Humble binaries (cv_bridge, tf2, PCL bindings) are
# compiled against the 1.x ABI and segfault against 2.x.
numpy>=1.24,<2.0
scipy>=1.10,<2.0
pyyaml>=6.0
tqdm>=4.66
rich>=13.7
loguru>=0.7

# --- Vision ----------------------------------------------------------------
# Do NOT install opencv-python on the Jetson - JetPack ships a CUDA-enabled cv2.
opencv-python>=4.8,<4.11
pillow>=10.2
pycocotools>=2.0.7
albumentations>=1.4          # detection-time augmentation

# --- Detection stack (YOLOX) ----------------------------------------------
# YOLOX itself is NOT on PyPI in a usable form - it is vendored via tita.repos
# into external/YOLOX and installed editable:
#     pip install -e external/YOLOX
# See perception/README.md.
#
# torch: PyPI only serves CPU-only wheels on Windows, and the CUDA wheels it
# serves on Linux (cu126) do not support Blackwell GPUs (RTX 50xx, sm_120).
# The cu128 builds from PyTorch's own index do. --extra-index-url keeps PyPI
# as the primary index for everything else; pip prefers the "+cu128" local
# version over the plain PyPI one.
--extra-index-url https://download.pytorch.org/whl/cu128
torch==2.8.0
torchvision==0.23.0
thop>=0.1.1                  # FLOP counting, required by YOLOX exps
tabulate>=0.9
psutil>=5.9
ninja>=1.11                  # YOLOX setup.py build dependency

# --- Model export / inference ---------------------------------------------
onnx>=1.15
onnxsim>=0.4.36
# onnxruntime-gpu needs CUDA 12 + cuDNN 9 DLLs at runtime. On the workstation
# they are borrowed from the torch cu128 wheel via onnxruntime.preload_dlls()
# (done in src/tita_perception/.../detectors/yolox_onnx.py). No separate CUDA
# toolkit install is needed. Jetson: see requirements-jetson.txt.
onnxruntime-gpu>=1.21        # >=1.21 for preload_dlls()

# --- Experiment tracking / evaluation -------------------------------------
tensorboard>=2.16
pandas>=2.2
matplotlib>=3.8
seaborn>=0.13

# --- Mapping / geometry (offline analysis of bags & maps) -----------------
open3d>=0.18                 # point-cloud inspection, map QA
transforms3d>=0.4
evo>=1.28                    # trajectory / SLAM benchmarking (ATE, RPE)
